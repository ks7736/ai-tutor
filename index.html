// 통신 성공률을 높인 Gemini API 호출 함수 async function callGeminiAPI(promptText, apiKey) { // 최신 v1beta REST API 엔드포인트 const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${apiKey}`; const response = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: promptText }] }] }) }); if (!response.ok) { const errorData = await response.json(); throw new Error(errorData.error?.message || `HTTP error! status: ${response.status}`); } const data = await response.json(); return data.candidates[0].content.parts[0].text; }
